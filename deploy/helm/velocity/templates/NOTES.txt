  HTTP API:
    export POD_NAME=$(kubectl get pods -n {{ .Release.Namespace }} \
      -l "app.kubernetes.io/name={{ include "velocity.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" \
      -o jsonpath="{.items[0].metadata.name}")

    kubectl port-forward $POD_NAME {{ .Values.service.httpPort }}:{{ .Values.service.httpPort }} -n {{ .Release.Namespace }}
    open http://127.0.0.1:{{ .Values.service.httpPort }}

  gRPC API:
    kubectl port-forward $POD_NAME {{ .Values.service.grpcPort }}:{{ .Values.service.grpcPort }} -n {{ .Release.Namespace }}
    # Connect your gRPC client to 127.0.0.1:{{ .Values.service.grpcPort }}
